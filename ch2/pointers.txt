func update(s []int) {

	fmt.Println("The length of a slice s is ", len(s))
	fmt.Println("The capacity of a slice s is ", cap(s))

	s[0] = 10
	s[1] = 20
	s[2] = 30

	fmt.Println("Printing a slice from the Update function before append ", s)

	s = append(s, 200, 300, 400, 500)

	fmt.Println("Printing a slice from the Update function after append ", s)

}
